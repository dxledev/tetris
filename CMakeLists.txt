cmake_minimum_required(VERSION 3.21)
project(Tetris VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets)

add_library(tetris_game src/core/game.cpp)
set_target_properties(tetris_game PROPERTIES AUTOMOC OFF)
target_include_directories(tetris_game PUBLIC src)

add_library(tetris_core src/core/theme.cpp src/core/theme_source.cpp src/core/storage.cpp)
target_include_directories(tetris_core PUBLIC src)
target_link_libraries(tetris_core PUBLIC tetris_game Qt6::Core Qt6::Gui)

add_library(tetris_ui
    src/ui/board.cpp src/ui/piece_preview.cpp src/ui/preferences.cpp src/ui/modal_backdrop.cpp
    src/ui/stats.cpp src/ui/theme_watcher.cpp src/ui/window.cpp)
target_link_libraries(tetris_ui PUBLIC tetris_core Qt6::Widgets)

add_executable(tetris src/main.cpp src/cli.cpp)
target_link_libraries(tetris PRIVATE tetris_ui)

foreach(target tetris_game tetris_core tetris_ui tetris)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(game_tests tests/game_tests.cpp)
    set_target_properties(game_tests PROPERTIES AUTOMOC OFF)
    target_include_directories(game_tests PRIVATE src/core)
    target_compile_options(game_tests PRIVATE -UNDEBUG)
    target_link_libraries(game_tests PRIVATE tetris_game)
    add_test(NAME game COMMAND game_tests)
    add_executable(theme_tests tests/theme_tests.cpp)
    target_link_libraries(theme_tests PRIVATE tetris_ui Qt6::Test)
    add_test(NAME theme COMMAND theme_tests)
    set_tests_properties(theme PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(source_tests tests/source_tests.cpp)
    target_link_libraries(source_tests PRIVATE tetris_core Qt6::Test)
    add_test(NAME sources COMMAND source_tests)
    add_executable(storage_tests tests/storage_tests.cpp)
    target_link_libraries(storage_tests PRIVATE tetris_core Qt6::Test)
    add_test(NAME storage COMMAND storage_tests)
    add_executable(ui_tests tests/ui_tests.cpp)
    target_link_libraries(ui_tests PRIVATE tetris_ui Qt6::Test)
    add_test(NAME ui COMMAND ui_tests $<TARGET_FILE:tetris>)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME cli COMMAND /usr/bin/bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh $<TARGET_FILE:tetris>)
    add_test(NAME install COMMAND /usr/bin/bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh)
endif()

include(GNUInstallDirs)
install(TARGETS tetris RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES assets/io.github.tetris.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/io.github.tetris.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
